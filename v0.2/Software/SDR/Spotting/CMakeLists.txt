cmake_minimum_required(VERSION 3.16)
project(wwu2026_sdr_harvester LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_path(SOAPYSDR_INCLUDE_DIR SoapySDR/Device.hpp REQUIRED)
find_library(SOAPYSDR_LIBRARY SoapySDR REQUIRED)
find_path(LIQUID_INCLUDE_DIR liquid/liquid.h REQUIRED)
find_library(LIQUID_LIBRARY liquid REQUIRED)
find_path(SNDFILE_INCLUDE_DIR sndfile.h REQUIRED)
find_library(SNDFILE_LIBRARY sndfile REQUIRED)
find_path(NLOHMANN_JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
add_executable(harvester harvester.cpp)
target_include_directories(harvester PRIVATE ${SOAPYSDR_INCLUDE_DIR} ${LIQUID_INCLUDE_DIR} ${SNDFILE_INCLUDE_DIR} ${NLOHMANN_JSON_INCLUDE_DIR})
target_link_libraries(harvester PRIVATE ${SOAPYSDR_LIBRARY} ${LIQUID_LIBRARY} ${SNDFILE_LIBRARY} pthread)
target_compile_options(harvester PRIVATE -Wall -Wextra -Wpedantic)
